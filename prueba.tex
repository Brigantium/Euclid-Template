	\section{Espacios Vectoriales}
	Comenzamos el estudio con el concepto de \gls{vector} y la idea de un \gls{espacio} vectorial \cite{anton}. 
	
	\begin{definition}[Espacio Vectorial]
		Un espacio vectorial sobre un cuerpo $K$ es un conjunto no vacío $V$ dotado de dos operaciones (suma y producto por un escalar) que cumplen ocho \glspl{axioma} fundamentales.
	\end{definition}

	\subsection{Independencia Lineal}
	Una \gls{base} se define a través de vectores independientes. 
	
	\begin{theorem}[Base Invariante]
		Todas las bases de un espacio vectorial finitamente generado tienen la misma \gls{cardinalidad} \cite{herstein}.
	\end{theorem}
	
	\begin{proof}
		Sea $B_1$ y $B_2$ dos bases del espacio $V$. Por el lema de intercambio de Steinitz, podemos sustituir elementos de $B_1$ en $B_2$ sin perder la propiedad de generar a $V$, lo que implica que $|B_1| \leq |B_2|$ y viceversa. Por tanto, $|B_1| = |B_2|$.
	\end{proof}

	\section{Límites y Continuidad}
	El \gls{calculo} se funda en el \gls{limite}. Una función goza de \gls{continuidad} si el límite coincide con el valor evaluado \cite{stewart}. 
	
	\begin{definition}[Límite]
		Sea $f: A \to \mathbb{R}$ una \gls{funcion}. Se dice que $\lim_{x \to c} f(x) = L$ si para todo $\epsilon > 0$ existe un $\delta > 0$ tal que si $0 < |x - c| < \delta$, entonces $|f(x) - L| < \epsilon$.
	\end{definition}
	
	\begin{figure}[h]

	\section{Derivadas}
	La \gls{derivada} mide el cambio. Relacionada con funciones como el \gls{seno} y el \gls{coseno}.
	
	\begin{lemma}[Regla de la Cadena]
		Si $g$ es derivable en $x$ y $f$ derivable en $g(x)$, entonces $f \circ g$ es derivable en $x$ \cite{rudin}.
	\end{lemma}

	\section{Integración}
	La \gls{integral} es el operador inverso\footnote{Bajo condiciones específicas establecidas en el Teorema Fundamental del Cálculo.}.
	
	\begin{theorem}[Teorema Fundamental del Cálculo]
		Sea $f$ continua en $[a,b]$. Entonces la función $F(x) = \int_a^x f(t)dt$ es derivable y $F'(x) = f(x)$. Citando a Apostol \cite{apostol}.
	\end{theorem}

	\section{Espacios Topológicos}
	La \gls{topologia} estudia conjuntos \cite{munkres}. Un \gls{espacio} topológico generaliza a los espacios métricos \cite{rudin}.
	
	\begin{definition}[Espacio Topológico]
		Un espacio topológico es un par $(X, \tau)$ donde $\tau$ es una colección de subconjuntos de $X$ que contiene al conjunto vacío y a $X$, y es cerrada bajo uniones arbitrarias e intersecciones finitas.
	\end{definition}

	\section{Varianza y Esperanza}
	La \gls{varianza} mide la dispersión. 
	
	\begin{theorem}[Desigualdad de Chebyshev]
		Para cualquier variable aleatoria $X$ con media $\mu$ y varianza $\sigma^2 > 0$, la probabilidad de que $X$ se desvíe de $\mu$ en más de $k\sigma$ es a lo sumo $1/k^2$. \cite{ross, spivak, strang}.
	\end{theorem}

# Ejemplo de código en R para calcular la varianza
calcular_varianza <- function(datos) {
	media <- mean(datos)
	varianza <- sum((datos - media)^2) / (length(datos) - 1)
	return(varianza)
}
